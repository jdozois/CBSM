let
    SiteUrl = "https://jdozois.github.io/CBSM/",
    Url = SiteUrl & (if Text.EndsWith(SiteUrl, "/") then "" else "/") & "data/changes.json",
    Source = Json.Document(Web.Contents(Url)),
    Row = (d, kind) => (c) => [
        Date = Date.From(DateTimeZone.FromText(d[date])),
        Change = kind,
        Title = c[title],
        Id = c[id]
    ],
    Rows = List.Combine(List.Transform(Source, (d) =>
        List.Transform(d[added], Row(d, "Added")) &
        List.Transform(d[changed], Row(d, "Edited")) &
        List.Transform(d[removed], Row(d, "Removed")))),
    Empty = #table(type table [Date = date, Change = text, Title = text, Id = text], {}),
    Result = if List.IsEmpty(Rows) then Empty
             else Table.TransformColumnTypes(Table.FromRecords(Rows),
                  {{"Date", type date}, {"Change", type text}, {"Title", type text}, {"Id", type text}})
in
    Result
